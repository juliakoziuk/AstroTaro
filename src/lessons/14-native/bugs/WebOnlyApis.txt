// ❌ Так працює у вебі, але НЕ в React Native
function Profile() {
  const saved = localStorage.getItem('name')        // немає localStorage
  const width = window.innerWidth                    // немає window
  document.title = 'Профіль'                         // немає document

  return (
    <div className="profile">                        {/* немає div і CSS-класів */}
      <h1>{saved}</h1>                               {/* немає h1 */}
      <button onClick={() => alert('Привіт')}>Hi</button>  {/* немає button, onClick */}
    </div>
  )
}
